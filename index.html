async function fetchNSEData() { // REPLACE the URL below with your actual Google Script Web App URL const googleScriptUrl = "YOUR_DEPLOYED_URL_HERE"; try { const response = await fetch(googleScriptUrl); const nseData = await response.json(); const tableBody = document.getElementById('tableBody'); tableBody.innerHTML = ''; nseData.data.forEach(stock => { const row = ` ${stock.symbol} ${stock.lastPrice} +${stock.pChange}% ${stock.tradedQuantity} `; tableBody.innerHTML += row; }); document.getElementById('status').style.display = 'none'; document.getElementById('nseTable').style.display = 'table'; } catch (error) { document.getElementById('status').innerHTML = "Error fetching data through Google Bridge."; console.error("Fetch error:", error); } }
